/* Physical launch trails, drag, gravity, layered explosions and smoke. */
const fireColors=[
 [222,195,139],
 [215,166,184],
 [120,171,145],
 [235,230,214],
 [215,167,118]
];

function launchFirework(){
 const depth=rand(.45,1);
 const x=rand(.1,.9)*screenW;
 rockets.push({
  x,y:screenH*1.04,
  vx:rand(-16,16),
  vy:-screenH*rand(.43,.63),
  ay:-screenH*.13,
  targetY:rand(.09,.43)*screenH,
  depth,
  trail:[],
  color:Math.floor(rand(0,fireColors.length))
 });
}

function explode(r){
 const count=Math.floor((reduced?27:68)*r.depth);
 const color=fireColors[r.color];
 const shape=Math.random();

 for(let i=0;i<count;i++){
  const angle=i/count*Math.PI*2+rand(-.05,.05);
  const ring=shape>.65?rand(.8,1):rand(.35,1);
  const velocity=rand(85,155)*r.depth*ring;
  const life=rand(1.35,2.6);

  fireParticles.push({
   x:r.x,y:r.y,
   vx:Math.cos(angle)*velocity,
   vy:Math.sin(angle)*velocity,
   life,maxLife:life,
   depth:r.depth,
   color,
   size:rand(.75,1.65)*r.depth,
   trail:[],
   sparkle:Math.random()>.8
  });
 }

 smoke.push({
  x:r.x,y:r.y,
  radius:rand(12,24)*r.depth,
  life:3.5,
  maxLife:3.5,
  depth:r.depth
 });

 if(fireParticles.length>700){
  fireParticles.splice(0,fireParticles.length-700);
 }
 if(smoke.length>12)smoke.shift();
}

function updateFireworks(dt){
 fireClock-=dt;

 if(coreAge>7.3&&coreAge<37&&fireClock<=0){
  launchFirework();
  if(!reduced&&Math.random()>.58)launchFirework();
  fireClock=reduced?rand(1.8,2.8):rand(.75,1.4);
 }else if(coreAge>=37&&fireClock<=0){
  launchFirework();
  fireClock=reduced?rand(4,6):rand(2.5,4.5);
 }

 for(let i=rockets.length-1;i>=0;i--){
  const r=rockets[i];
  r.trail.push({x:r.x,y:r.y});
  if(r.trail.length>17)r.trail.shift();

  r.vy+=r.ay*dt;
  r.x+=r.vx*dt;
  r.y+=r.vy*dt;

  if(r.y<=r.targetY){
   explode(r);
   rockets.splice(i,1);
  }
 }

 for(let i=fireParticles.length-1;i>=0;i--){
  const p=fireParticles[i];
  p.life-=dt;
  if(p.life<=0){
   fireParticles.splice(i,1);
   continue;
  }

  p.trail.push({x:p.x,y:p.y});
  if(p.trail.length>6)p.trail.shift();

  const drag=Math.exp(-dt*.62);
  p.vx*=drag;
  p.vy=p.vy*drag+26*p.depth*dt;
  p.x+=p.vx*dt;
  p.y+=p.vy*dt;
 }

 for(let i=smoke.length-1;i>=0;i--){
  const s=smoke[i];
  s.life-=dt;
  s.radius+=dt*13*s.depth;
  s.x+=dt*7*s.depth;
  s.y-=dt*3;
  if(s.life<=0)smoke.splice(i,1);
 }
}

function drawFireworks(){
 fctx.clearRect(0,0,screenW,screenH);
 const reveal=clamp((coreAge-1.2)/3,0,1);

 stars.forEach(s=>{
  const twinkle=.55+Math.sin(coreAge*.65+s.phase)*.2;
  fctx.globalAlpha=reveal*twinkle*s.depth;
  fctx.fillStyle='#e1dfcb';

  const x=s.x*screenW+pointer.x*s.depth*9;
  const y=s.y*screenH+pointer.y*s.depth*6;
  fctx.beginPath();
  fctx.arc(x,y,s.r,0,Math.PI*2);
  fctx.fill();
 });
 fctx.globalAlpha=1;

 if(coreAge>5){
  const opacity=clamp((coreAge-5)/4,0,1);
  for(let i=0;i<(reduced?4:9);i++){
   const phase=i*2.399;
   const x=screenW*((i*.137+.08)%1)+Math.sin(coreAge*.17+phase)*18;
   const y=screenH*(1-((coreAge*.012+i*.119)%1));
   const size=8+(i%3)*4;

   fctx.save();
   fctx.translate(x,y);
   fctx.rotate(Math.sin(coreAge*.2+phase)*.22);
   fctx.globalAlpha=opacity*(.08+(i%3)*.025);
   fctx.strokeStyle=i%2?'#d4aab3':'#c1cda9';
   fctx.lineWidth=.8;
   fctx.beginPath();
   fctx.moveTo(0,size*.7);
   fctx.bezierCurveTo(-size*1.2,-size*.05,-size*.65,-size*.9,0,-size*.35);
   fctx.bezierCurveTo(size*.65,-size*.9,size*1.2,-size*.05,0,size*.7);
   fctx.stroke();
   fctx.restore();
  }
 }

 smoke.forEach(s=>{
  const alpha=Math.sin(Math.PI*clamp(s.life/s.maxLife,0,1))*.035;
  const g=fctx.createRadialGradient(s.x,s.y,0,s.x,s.y,s.radius);
  g.addColorStop(0,`rgba(172,181,164,${alpha})`);
  g.addColorStop(1,'rgba(172,181,164,0)');
  fctx.fillStyle=g;
  fctx.fillRect(s.x-s.radius,s.y-s.radius,s.radius*2,s.radius*2);
 });

 rockets.forEach(r=>{
  const color=fireColors[r.color];
  for(let j=1;j<r.trail.length;j++){
   const a=r.trail[j-1],b=r.trail[j];
   fctx.strokeStyle=`rgba(${color.join(',')},${j/r.trail.length*.65})`;
   fctx.lineWidth=Math.max(.65,r.depth*1.35);
   fctx.beginPath();
   fctx.moveTo(a.x,a.y);
   fctx.lineTo(b.x,b.y);
   fctx.stroke();
  }
  fctx.fillStyle='#f0e4c8';
  fctx.fillRect(r.x-1,r.y-2,1.5,3);
 });

 fireParticles.forEach(p=>{
  const life=clamp(p.life/p.maxLife,0,1);
  const opacity=Math.pow(life,.75);
  const flicker=p.sparkle?.72+.28*Math.sin(coreAge*28+p.x):1;

  for(let j=1;j<p.trail.length;j++){
   const a=p.trail[j-1],b=p.trail[j];
   fctx.strokeStyle=`rgba(${p.color.join(',')},${opacity*flicker*j/p.trail.length*.5})`;
   fctx.lineWidth=Math.max(.45,p.size*.75);
   fctx.beginPath();
   fctx.moveTo(a.x,a.y);
   fctx.lineTo(b.x,b.y);
   fctx.stroke();
  }

  fctx.globalAlpha=opacity*flicker;
  fctx.fillStyle=`rgb(${p.color.join(',')})`;
  fctx.fillRect(p.x-p.size/2,p.y-p.size/2,p.size,p.size);
 });

 fctx.globalAlpha=1;
}

function updateCore(dt){
 if(!coreUnlocked()||status!=='ending')return;
 coreAge+=dt;

 if(coreAge>.65&&!$('#ending').classList.contains('awaken')){
  $('#ending').classList.add('awaken');
 }

 corePhotos.forEach((photo,i)=>{
  if(coreAge>3.1+i*.85){
   photo.classList.add('revealed');
  }
  if(!reduced&&coreAge>7+i*.85){
   photo.style.setProperty('--float',Math.sin(coreAge*.35+i*1.7)*8+'px');
  }
 });

 if(coreAge>7.5&&!coreSoundPlayed){
  coreSoundPlayed=true;
  tone('final');
 }

 if(coreAge>8.1&&!coreTitleShown){
  coreTitleShown=true;
  $('#core-copy').classList.add('revealed');
  rainStart=ambientTime;
  words.forEach(w=>w.y=rand(-.8,-.05));
 }

 if(coreAge>16){
  const signature=$('#creator-signature');
  if(signature.hidden){
   signature.hidden=false;
   void signature.offsetWidth;
   signature.classList.add('revealed');
  }
 }

 const cameraX=reduced?0:Math.sin(coreAge*.16)*2.2+pointer.x*2;
 const cameraY=reduced?0:Math.cos(coreAge*.13)*1.3-pointer.y*1.2;
 $('#ending').style.setProperty('--camera-x',cameraX+'deg');
 $('#ending').style.setProperty('--camera-y',cameraY+'deg');

 updateFireworks(dt);
 drawFireworks();
}

function drawAtmosphere(dt){
 actx.clearRect(0,0,screenW,screenH);
 const inArcade=entered&&mode==='arcade'&&status!=='ending';
 const age=rainStart===null?0:ambientTime-rainStart;
 const warp=entered&&rainStart!==null&&age<1.8?(1.8-age)*2.5:0;

 dust.forEach(p=>{
  if(!reduced){
   const slow=status==='reward'?.35:1;
   p.y-=dt*p.speed*(1+warp)*slow;
   p.x+=Math.sin(ambientTime*.15+p.phase)*dt*.002;
   if(p.y<-.02)p.y=1.02;
   if(p.x>1.02)p.x=-.02;
   if(p.x<-.02)p.x=1.02;
  }

  const x=p.x*screenW+pointer.x*p.r*8;
  const y=p.y*screenH+pointer.y*p.r*8;
  const opacity=inArcade&&!gate?.1:.17+p.r*.09;
  actx.fillStyle=`rgba(225,224,193,${opacity})`;
  actx.beginPath();
  actx.arc(x,y,p.r*(1+warp*.12),0,Math.PI*2);
  actx.fill();
 });

 if(rainStart===null||inArcade)return;

 let blend=clamp((age-3)/5,0,1);
 blend=blend*blend*(3-2*blend);
 const color=[38,111,77].map((v,i)=>Math.round(lerp(v,[216,166,179][i],blend)));

 words.forEach(w=>{
  if(!reduced){
   w.y+=dt*w.speed;
   if(w.y>1.14){
    w.y=-.15;
    w.x=rand(.025,.9);
   }
  }

  actx.save();
  actx.translate(
   w.x*screenW+Math.sin(ambientTime*.23+w.size)*12+pointer.x*w.depth*9,
   w.y*screenH
  );
  actx.rotate(w.angle+Math.sin(ambientTime*.2)*.025);
  actx.scale(.86+w.depth*.14,.75+w.depth*.25);
  actx.font=`italic ${w.size}px Georgia`;
  actx.fillStyle=`rgba(${color.join(',')},${.12+w.depth*.18})`;
  actx.fillText(w.text,0,0);
  actx.restore();
 });
}

  rafId=requestAnimationFrame(frame);
 }

function frame(timestamp){
 rafId=0;
 if(document.hidden||pageSuspended)return;

 const dt=previousFrame?Math.min((timestamp-previousFrame)/1000,.05):0;
 previousFrame=timestamp;
 ambientTime+=dt;

 if(resizePending){
  resizeAll();
 }

 const currentLandscape=isLandscape();
 if(currentLandscape!==landscape){
  landscape=currentLandscape;
  if(entered&&status!=='ending'&&landscape&&!lastPhysical){
   setMode('arcade');
  }
  lastPhysical=landscape;
  syncAccess();
  resizePending=true;
 }

 updateMusicFade(dt);
 musicUIClock+=dt;
 if(musicUIClock>1){
  musicUIClock=0;
  updateSound();
 }

 drawAtmosphere(dt);

 if(status==='ending'){
  avatar.hidden=true;
  updateCore(dt);
 }else if(entered&&mode==='arcade'&&landscape&&!gate&&!rotationSettling){
  if(!pausedByUser){
   updateArcade(dt);
  }

  if(status==='ending'){
   updateCore(0);
  }else{
   if(worldDirty){
    drawGame();
   }

   if(status==='reward'||status==='transition'){
    avatar.hidden=true;
    rewardDrawClock+=dt;
    if(rewardDrawClock>=1/30){
     rewardDrawClock=0;
     drawRewardWind();
    }
   }else if(status==='playing'){
    renderAvatar();
   }else{
    avatar.hidden=true;
   }
  }
 }else{
  avatar.hidden=true;
 }

 if(entered&&mode==='story'&&status!=='ending'&&!reduced){
  scrollSmooth+=(scrollTarget-scrollSmooth)*(1-Math.exp(-dt*8));
  parallaxElements.forEach(el=>{
   const section=el.closest('.story-section');
   const local=scrollSmooth-section.offsetTop;
   if(local>-screenH&&local<section.offsetHeight){
    el.style.transform=`translate3d(0,${local*Number(el.dataset.parallax)}px,0)`;
   }
  });
 }

 requestFrame();
}

/* Initialization occurs once. A full document reload is the only reset. */
landscape=isLandscape();
lastPhysical=landscape;
resizeAll();
resetLevel();
showBriefing();
updateProgression();
syncAccess();
drawGame();
updateSound();
loadAvatar();

/* Audible autoplay is attempted now. If blocked, natural interaction retries
   without restarting the track or interrupting the experience. */
attemptMusic();
requestFrame();

})();
